**文件**：`src/types/game.ts`

```typescript
// ============================================================
// 折叠相关
// ============================================================

export type FoldType =
  | 'horizontal'      // 横向对折（上下合拢）
  | 'vertical'        // 纵向对折（左右合拢）
  | 'diagonal-left'   // 左上-右下对角折叠
  | 'diagonal-right'  // 右上-左下对角折叠
  | 'triangle'        // 三角折叠（60° 三等分）
  | 'quarter'         // 四折（四分之一）
  | 'eighth'          // 八折（八分之一）

export interface FoldOption {
  id: string
  name: string                // "横向对折"
  type: FoldType
  icon: string                // 内联 SVG 字符串（折叠方式的小图标）
  description: string         // "纸张上下合拢，展开后左右镜像对称"
}

// ============================================================
// 剪裁相关
// ============================================================

export interface CutOption {
  id: string
  name: string                // "剪掉右上角"
  icon: string                // 内联 SVG 字符串（剪裁形状的小图标）
  // 在折叠后纸张上的剪裁区域（SVG path 的 d 属性）
  // 用于在 PaperCanvas 上高亮预览
  foldedRegion: string
  // 展开后的完整镂空区域（SVG path 的 d 属性）
  // 这是预计算好的结果，直接叠加到最终图案上
  unfoldedRegion: string
}

// ============================================================
// 关卡相关
// ============================================================

// 单步操作 = 选一个折叠 + 选一个剪裁
export interface LevelStep {
  foldOptions: FoldOption[]
  cutOptions: CutOption[]
}

export interface Level {
  id: string                  // "1-1"
  chapter: number             // 1
  order: number               // 1
  name: string                // "窗花"
  description: string         // "学习基础折叠，剪出对称图形"
  difficulty: 1 | 2 | 3 | 4 | 5
  // 目标纹样（完整的 SVG 内部标记，用于 TargetPreview 渲染）
  targetSVG: string
  // 操作步骤列表（1-3 步）
  steps: LevelStep[]
  // 提示文本（3 级）
  hints: string[]
}

// ============================================================
// 游戏状态
// ============================================================

export type GameMode = 'level' | 'challenge' | 'free'

export interface SelectedStep {
  fold: FoldOption
  cut: CutOption
}

export type GamePhase =
  | 'selecting-fold'         // 正在选择折叠方式
  | 'fold-animating'         // 折叠动画播放中
  | 'selecting-cut'          // 正在选择剪裁位置
  | 'cut-animating'          // 剪裁动画播放中
  | 'ready-to-unfold'        // 可以展开
  | 'unfolding'              // 展开动画播放中
  | 'scored'                 // 已评分

export interface GamePlayState {
  level: Level
  currentStepIndex: number   // 当前操作步骤（0-based）
  selectedSteps: SelectedStep[]  // 已完成的步骤
  phase: GamePhase
  score: number | null       // 评分（0-100）
  stars: number | null       // 星级（1-5）
}

// ============================================================
// 章节
// ============================================================

export interface Chapter {
  id: number
  name: string
  subtitle: string
  levels: Level[]
}
```

### 4.3 预计算数据生成说明

**每个关卡的 `unfoldedRegion` 必须预计算**。计算方法：

1. 在纸上画出折叠方式
2. 在折叠后的纸上标出剪裁区域
3. 在脑中（或用工具）将剪裁区域按折叠逆序展开
4. 将展开后的区域转换为 SVG path 的 `d` 属性

**AI 执行时**：AI 需要根据每个关卡的折叠方式和剪裁位置，预计算出展开后的 SVG path。这不需要运行时代码，只需要在编写关卡数据时一次性计算好。

## 五、游戏状态管理

**文件**：`src/game/hooks/useGamePlay.ts`

```typescript
import { useState, useCallback } from 'react'
import {
  GamePlayState,
  GamePhase,
  Level,
  FoldOption,
  CutOption,
  SelectedStep,
} from '@/types/game'

export function useGamePlay(level: Level) {
  const [state, setState] = useState<GamePlayState>({
    level,
    currentStepIndex: 0,
    selectedSteps: [],
    phase: 'selecting-fold',
    score: null,
    stars: null,
  })

  const currentStep = level.steps[state.currentStepIndex]
  const isLastStep = state.currentStepIndex >= level.steps.length - 1

  // 选择折叠
  const selectFold = useCallback((fold: FoldOption) => {
    setState(prev => ({
      ...prev,
      phase: 'fold-animating',
    }))

    // 折叠动画结束后切换到选择剪裁
    setTimeout(() => {
      setState(prev => ({
        ...prev,
        phase: 'selecting-cut',
        // 暂存选中的折叠（未确认剪裁前不存入 selectedSteps）
        _pendingFold: fold,
      }))
    }, 500)
  }, [])

  // 选择剪裁
  const selectCut = useCallback((cut: CutOption) => {
    setState(prev => {
      const pendingFold = (prev as any)._pendingFold as FoldOption
      if (!pendingFold) return prev

      const newStep: SelectedStep = { fold: pendingFold, cut }
      const newSelectedSteps = [...prev.selectedSteps, newStep]
      const isLast = prev.currentStepIndex >= level.steps.length - 1

      return {
        ...prev,
        selectedSteps: newSelectedSteps,
        phase: isLast ? 'ready-to-unfold' : 'cut-animating',
        _pendingFold: undefined,
      }
    })

    // 如果不是最后一步，剪裁动画结束后进入下一步
    setTimeout(() => {
      setState(prev => {
        if (prev.phase === 'ready-to-unfold') return prev
        return {
          ...prev,
          currentStepIndex: prev.currentStepIndex + 1,
          phase: 'selecting-fold',
        }
      })
    }, 300)
  }, [level.steps.length])

  // 展开
  const unfold = useCallback(() => {
    setState(prev => ({ ...prev, phase: 'unfolding' }))

    // 展开动画结束后评分
    setTimeout(() => {
      // 计算评分（将在第六节实现）
      const resultSVG = buildResultSVG(prev.selectedSteps)
      const { score, stars } = calculateScore(prev.level.targetSVG, resultSVG)

      setState(prev => ({
        ...prev,
        phase: 'scored',
        score,
        stars,
      }))
    }, 400 * level.steps.length)  // 每步 400ms 展开动画
  }, [level])

  // 重试
  const retry = useCallback(() => {
    setState({
      level,
      currentStepIndex: 0,
      selectedSteps: [],
      phase: 'selecting-fold',
      score: null,
      stars: null,
    })
  }, [level])

  // 使用提示
  const useHint = useCallback((hintIndex: number) => {
    // 直接展示提示文本（alert 或 UI 内显示）
    const hint = level.hints[hintIndex]
    if (hint) alert(hint)
  }, [level])

  return {
    state,
    currentStep,
    isLastStep,
    selectFold,
    selectCut,
    unfold,
    retry,
    useHint,
  }
}

/**
 * 根据已选步骤构建结果 SVG
 * 核心逻辑：将每步的 unfoldedRegion 叠加
 */
function buildResultSVG(steps: SelectedStep[]): string {
  return steps
    .map(step => `<path d="${step.cut.unfoldedRegion}" fill="white" />`)
    .join('\n')
}
```

## 六、评分计算

**文件**：`src/lib/score.ts`

```typescript
/**
 * 使用离屏 Canvas 计算两个 SVG 的像素重合度（IoU）
 */
export function calculateScore(
  targetSVG: string,
  resultSVG: string,
): { score: number; stars: number } {
  const SIZE = 100  // 与 viewBox 一致
  const canvas = document.createElement('canvas')
  canvas.width = SIZE
  canvas.height = SIZE
  const ctx = canvas.getContext('2d')!

  // 1. 绘制目标 SVG → 获取像素
  ctx.clearRect(0, 0, SIZE, SIZE)
  ctx.fillStyle = '#000'
  ctx.fillRect(0, 0, SIZE, SIZE)
  ctx.fillStyle = '#fff'
  drawSVGToCanvas(ctx, targetSVG, SIZE)
  const targetPixels = ctx.getImageData(0, 0, SIZE, SIZE).data

  // 2. 绘制结果 SVG → 获取像素
  ctx.clearRect(0, 0, SIZE, SIZE)
  ctx.fillStyle = '#000'
  ctx.fillRect(0, 0, SIZE, SIZE)
  ctx.fillStyle = '#fff'
  drawSVGToCanvas(ctx, resultSVG, SIZE)
  const resultPixels = ctx.getImageData(0, 0, SIZE, SIZE).data

  // 3. 计算 IoU
  let intersection = 0
  let union = 0

  for (let i = 0; i < targetPixels.length; i += 4) {
    const isTarget = targetPixels[i] > 128  // 白色像素
    const isResult = resultPixels[i] > 128

    if (isTarget && isResult) intersection++
    if (isTarget || isResult) union++
  }

  const accuracy = union > 0 ? intersection / union : 0
  const score = Math.round(accuracy * 100)

  // 4. 星级
  let stars = 1
  if (accuracy >= 0.95) stars = 5
  else if (accuracy >= 0.85) stars = 4
  else if (accuracy >= 0.70) stars = 3
  else if (accuracy >= 0.50) stars = 2

  return { score, stars }
}

/**
```
```typescript
 * 使用 Path2D 从 SVG path 数据绘制（更可靠）
 */
export function calculateScore(
  targetPaths: string[],   // 目标 SVG 的 path d 属性数组
  resultPaths: string[],   // 结果 SVG 的 path d 属性数组
): { score: number; stars: number } {
  const SIZE = 100
  const canvas = document.createElement('canvas')
  canvas.width = SIZE
  canvas.height = SIZE
  const ctx = canvas.getContext('2d')!

  // 绘制目标
  ctx.fillStyle = '#000'
  ctx.fillRect(0, 0, SIZE, SIZE)
  ctx.fillStyle = '#fff'
  targetPaths.forEach(d => {
    const path = new Path2D(d)
    ctx.fill(path)
  })
  const targetData = ctx.getImageData(0, 0, SIZE, SIZE).data

  // 绘制结果
  ctx.fillStyle = '#000'
  ctx.fillRect(0, 0, SIZE, SIZE)
  ctx.fillStyle = '#fff'
  resultPaths.forEach(d => {
    const path = new Path2D(d)
    ctx.fill(path)
  })
  const resultData = ctx.getImageData(0, 0, SIZE, SIZE).data

  // 计算 IoU
  let intersection = 0
  let union = 0
  for (let i = 0; i < targetData.length; i += 4) {
    const t = targetData[i] > 128
    const r = resultData[i] > 128
    if (t && r) intersection++
    if (t || r) union++
  }

  const accuracy = union > 0 ? intersection / union : 0
  const score = Math.round(accuracy * 100)

  let stars = 1
  if (accuracy >= 0.95) stars = 5
  else if (accuracy >= 0.85) stars = 4
  else if (accuracy >= 0.70) stars = 3
  else if (accuracy >= 0.50) stars = 2

  return { score, stars }
}
```

## 十一、关键注意事项

### 11.1 预计算数据的准确性

**这是整个游戏成败的关键**。`unfoldedRegion` 必须准确反映"在指定折叠方式下剪裁后展开"的真实结果。

**验证方法**：
1. 在纸上实际折叠并剪裁
2. 展开后拍照/扫描
3. 将展开结果与 SVG path 对比
4. 或使用在线 SVG 编辑器手动绘制展开结果

**AI 执行时的替代方案**：
- 如果 AI 无法精确预计算，可以先用简单的几何形状（圆形、三角形、矩形）作为关卡目标
- 这些简单形状的展开结果容易预计算
- 后续再逐步替换为复杂的真实剪纸纹样

### 11.2 不需要实现的功能

| 功能 | 原因 |
|------|------|
| 几何引擎（reflectPoint 等） | 所有结果预计算，不需要实时计算 |
| Canvas 实时渲染 | 使用 SVG 渲染，更简单 |
| 物理引擎 | 无物理模拟 |
| 后端 API | 纯前端，数据硬编码 |
| 用户系统 | 无登录/注册 |
| 排行榜 | 无后端存储 |

### 11.3 性能注意事项

| 关注点 | 方案 |
|--------|------|
| SVG 叠加数量 | 每关最多 3 步 × 1 个 path = 3 个 path 叠加，性能无忧 |
| Canvas 评分 | 仅在展开时执行一次，100×100 像素，毫秒级完成 |
| CSS 动画 | 使用 transform/opacity，GPU 加速 |
| 关卡数据 | 30 关数据量极小，无需代码分割 |
